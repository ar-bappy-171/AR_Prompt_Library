   174→   174→function getNextSerialNumber() {
   175→   175→    const maxSerial = state.prompts.reduce((max, prompt) => {
   176→   176→        const serial = Number.isInteger(prompt.serialNumber) && prompt.serialNumber > 0 ? prompt.serialNumber : 0;
   177→   177→        return Math.max(max, serial);
   178→   178→    }, 0);
   179→   179→    return maxSerial + 1;
   180→   180→}

   182→   182→function migratePromptSerialNumbers() {
   183→   183→    let nextSerial = getNextSerialNumber();
   184→   184→    state.prompts.forEach(prompt => {
   185→   185→        if (!Number.isInteger(prompt.serialNumber) || prompt.serialNumber < 1) {
   186→   186→            prompt.serialNumber = nextSerial++;
   187→   187→        }
   188→   188→    });
   189→   189→    saveState();
   190→   190→}

   488→   488→function handleFileImport(event) {
   489→   489→    const file = event.target.files[0];
   490→   490→    if (!file) return;
   491→   491→    
   492→   492→    if (!file.name.endsWith('.json')) {
   493→   493→        showToast('Please select a JSON file', 'error');
   494→   494→        return;
   495→   495→    }
   496→   496→    
   497→   497→    const reader = new FileReader();
   498→   498→    
   499→   499→    reader.onload = function(e) {
   500→   500→        try {
   501→   501→            const importedData = JSON.parse(e.target.result);
   502→   502→            
   503→   503→            // Validate imported data
   504→   504→            if (!importedData.prompts || !Array.isArray(importedData.prompts)) {
