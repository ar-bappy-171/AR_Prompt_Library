   127→            categories.forEach(cat => state.categories.set(cat.id, cat));

   132→                    id,

   134→                    color: data.color,
   135→                    icon: data.icon,
   136→                    count: 0,
   137→                    createdAt: new Date().toISOString()

   139→            });
   140→            saveCategories();

   142→

   144→        if (state.prompts.length === 0) {
   145→            addSampleData();
   146→        }

   148→        // Load custom order

   151→            state.customOrder = JSON.parse(customOrderData);

   160→            if (!state.customOrder.includes(p.id)) {
   161→                state.customOrder.push(p.id);
   162→            }
   163→        });
   164→
   165→        // Ensure every prompt has a serial number

   168→        console.log('📦 State loaded successfully');
   169→    } catch (error) {
   170→        console.error('❌ Failed to load state:', error);
   171→    }

   174→function getNextSerialNumber() {
   175→    const maxSerial = state.prompts.reduce((max, prompt) => {
   176→        const serial = Number.isInteger(prompt.serialNumber) && prompt.serialNumber > 0 ? prompt.serialNumber : 0;
   177→        return Math.max(max, serial);
   178→    }, 0);
   179→    return maxSerial + 1;
   180→}

   488→function handleFileImport(event) {
   489→    const file = event.target.files[0];
   490→    if (!file) return;
   491→    
   492→    if (!file.name.endsWith('.json')) {
   493→        showToast('Please select a JSON file', 'error');
   494→        return;
   495→    }
   496→    
   497→    const reader = new FileReader();
   498→    
   499→    reader.onload = function(e) {
   500→        try {
   501→            const importedData = JSON.parse(e.target.result);
   502→            
   503→            // Validate imported data
   504→            if (!importedData.prompts || !Array.isArray(importedData.prompts)) {
   505→                throw new Error('Invalid file format');
   506→            }
   507→            
   508→            showConfirmModal(
   509→                'Import Data',
   510→                `This will import ${importedData.prompts.length} prompts. Your current data will be preserved. Continue?`,
   511→                () => {
   512→                    // Merge prompts
   513→                    importedData.prompts.forEach(prompt => {
   514→                        // Check if prompt already exists
   515→                        const exists = state.prompts.some(p => p.id === prompt.id);
   516→                        if (!exists) {
   517→                            state.prompts.push(prompt);
   518→                        }
   519→                    });
   520→                    
   521→                    // Merge templates if they exist
   522→                    if (importedData.templates && Array.isArray(importedData.templates)) {
   523→                        importedData.templates.forEach(template => {
   524→                            const exists = state.templates.some(t => t.id === template.id);
   525→                            if (!exists) {
   526→                                state.templates.push(template);
   527→                            }
   528→                        });
   529→                    }
   530→                    
   531→                    // Merge categories if they exist
   532→                    if (importedData.categories && Array.isArray(importedData.categories)) {
   533→                        importedData.categories.forEach(category => {
   534→                            if (!state.categories.has(category.id)) {
   535→                                state.categories.set(category.id, category);
   536→                            }
   537→                        });
   538→                    }
   539→                    
   540→                    // Save state
   541→                    saveState();
   542→                    saveCategories();
   543→                    
   544→                    // Update UI
   545→                    updateCategoryCounts();
   546→                    updateStats();
   547→                    renderPrompts();
   548→                    initializeCategoriesDropdown();
   549→                    
   550→                    showToast(`Successfully imported ${importedData.prompts.length} prompts!`, 'success');
   551→                    
   552→                    // Reset file input
   553→                    event.target.value = '';
   554→                }
   555→            );
   556→        } catch (error) {
   557→            console.error('Import error:', error);
   558→            showToast('Error importing file. Please check the file format.', 'error');
   559→        }
   560→    };
   561→    
   562→    reader.readAsText(file);
   563→}
