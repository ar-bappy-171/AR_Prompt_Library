  1010→                        label.textContent = rating === 0 ? 'Not rated' : `${rating}/5`;
  1011→                    }
  1012→                });
  1013→                
  1014→                star.addEventListener('mouseover', function() {
  1015→                    const rating = parseInt(this.dataset.value);
  1016→                    stars.forEach((s, index) => {
  1017→                        if (index < rating) {
  1018→                            s.innerHTML = '<i class="fas fa-star"></i>';
  1019→                        } else {
  1020→                            s.innerHTML = '<i class="far fa-star"></i>';
  1021→                        }
  1022→                    });
  1023→                });
  1024→                
  1025→                star.addEventListener('mouseout', function() {
  1026→                    const currentRating = parseInt(document.getElementById('promptRating').value) || 0;
  1027→                    stars.forEach((s, index) => {
  1028→                        if (index < currentRating) {
  1029→                            s.innerHTML = '<i class="fas fa-star"></i>';
  1030→                        } else {
  1031→                            s.innerHTML = '<i class="far fa-star"></i>';
  1032→                        }

  1035→            });
  1036→            
  1037→            // Setup advanced section toggle
  1038→            const toggleHeader = document.getElementById('toggleAdvanced');
  1039→            if (toggleHeader) {
  1040→                toggleHeader.addEventListener('click', function() {
  1041→                    const content = document.getElementById('advancedContent');
  1042→                    const icon = this.querySelector('i');
  1043→
