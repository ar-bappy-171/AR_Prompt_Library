    94→function loadState() {
    95→    try {
    96→        // Load settings
    97→        const settings = localStorage.getItem(STORAGE_KEYS.SETTINGS);
    98→        if (settings) {
    99→            state.settings = { ...state.settings, ...JSON.parse(settings) };
   100→        }
   101→        
   102→        // Apply theme
   103→        document.documentElement.setAttribute('data-theme', state.settings.theme);
   104→        
   105→        // Load prompts
   106→        const promptsData = localStorage.getItem(STORAGE_KEYS.PROMPTS);
   107→        if (promptsData) {
   108→            state.prompts = JSON.parse(promptsData);
   109→        }
   110→        
   111→        // Load favorites
   112→        const favoritesData = localStorage.getItem(STORAGE_KEYS.FAVORITES);
   113→        if (favoritesData) {
   114→            state.favorites = JSON.parse(favoritesData);
   115→        }
   116→        
   117→        // Load templates
   118→        const templatesData = localStorage.getItem(STORAGE_KEYS.TEMPLATES);
   119→        if (templatesData) {
   120→            state.templates = JSON.parse(templatesData);
   121→        }
   122→        
   123→        // Load categories
   124→        const categoriesData = localStorage.getItem(STORAGE_KEYS.CATEGORIES);
   125→        if (categoriesData) {
   126→            const categories = JSON.parse(categoriesData);
   127→            categories.forEach(cat => state.categories.set(cat.id, cat));
   128→        } else {
   129→            // Initialize default categories
   130→            Object.entries(DEFAULT_CATEGORIES).forEach(([id, data]) => {
   131→                state.categories.set(id, {
   132→                    id,
   133→                    name: data.name,
   134→                    color: data.color,
   135→                    icon: data.icon,
   136→                    count: 0,
   137→                    createdAt: new Date().toISOString()
   138→                });
   139→            });
   140→            saveCategories();
   141→        }
   142→        
   143→        // Add sample data if empty
   144→        if (state.prompts.length === 0) {
   145→            addSampleData();
   146→        }
   147→        
   148→        // Load custom order
   149→        const customOrderData = localStorage.getItem(STORAGE_KEYS.CUSTOM_ORDER);
   150→        if (customOrderData) {
   151→            state.customOrder = JSON.parse(customOrderData);
   152→        } else {
   153→            // Initialize custom order from current prompts array order
   154→            state.customOrder = state.prompts.map(p => p.id);
   155→            saveCustomOrder();
   156→        }
   157→        
   158→        // Ensure all prompts are in the custom order
   159→        state.prompts.forEach(p => {
   160→            if (!state.customOrder.includes(p.id)) {
   161→                state.customOrder.push(p.id);
   162→            }
   163→        });
   164→
   165→        // Ensure every prompt has a serial number
   166→        migratePromptSerialNumbers();
   167→        
   168→        console.log('📦 State loaded successfully');
   169→    } catch (error) {
   170→        console.error('❌ Failed to load state:', error);
   171→    }
   172→}

   174→function getNextSerialNumber() {
   175→    const maxSerial = state.prompts.reduce((max, prompt) => {
   176→        const serial = Number.isInteger(prompt.serialNumber) && prompt.serialNumber > 0 ? prompt.serialNumber : 0;
   177→        return Math.max(max, serial);
   178→    }, 0);
   179→    return maxSerial + 1;
   180→}

   182→function migratePromptSerialNumbers() {
   183→    let nextSerial = getNextSerialNumber();
   184→    state.prompts.forEach(prompt => {
   185→        if (!Number.isInteger(prompt.serialNumber) || prompt.serialNumber < 1) {
   186→            prompt.serialNumber = nextSerial++;
   187→        }
   188→    });
   189→    saveState();
   190→}

   488→function handleFileImport(event) {
   489→    const file = event.target.files[0];
   490→    if (!file) return;
   491→    
   492→    if (!file.name.endsWith('.json')) {
   493→        showToast('Please select a JSON file', 'error');
   494→        return;
   495→    }
   496→    
   497→    const reader = new FileReader();
   498→    
   499→    reader.onload = function(e) {
   500→        try {
   501→            const importedData = JSON.parse(e.target.result);
   502→            
   503→            // Validate imported data
   504→            if (!importedData.prompts || !Array.isArray(importedData.prompts)) {
   505→                throw new Error('Invalid file format');
   506→            }
   507→            
   508→            showConfirmModal(
   509→                'Import Data',
   510→                `This will import ${importedData.prompts.length} prompts. Your current data will be preserved. Continue?`,
   511→                () => {
   512→                    // Merge prompts
   513→                    importedData.prompts.forEach(prompt => {
   514→                        // Check if prompt already exists
   515→                        const exists = state.prompts.some(p => p.id === prompt.id);
   516→                        if (!exists) {
   517→                            state.prompts.push(prompt);
   518→                        }
   519→                    });
   520→                    
   521→                    // Merge templates if they exist
   522→                    if (importedData.templates && Array.isArray(importedData.templates)) {
   523→                        importedData.templates.forEach(template => {
   524→                            const exists = state.templates.some(t => t.id === template.id);
   525→                            if (!exists) {
   526→                                state.templates.push(template);
   527→                            }
   528→                        });
   529→                    }
   530→                    
   531→                    // Merge categories if they exist
   532→                    if (importedData.categories && Array.isArray(importedData.categories)) {
   533→                        importedData.categories.forEach(category => {
   534→                            if (!state.categories.has(category.id)) {
   535→                                state.categories.set(category.id, category);
   536→                            }
   537→                        });
   538→                    }
   539→                    
   540→                    // Save state
   541→                    saveState();
   542→                    saveCategories();
   543→                    
   544→                    // Update UI
   545→                    updateCategoryCounts();
   546→                    updateStats();
   547→                    renderPrompts();
   548→                    initializeCategoriesDropdown();
   549→                    
   550→                    showToast(`Successfully imported ${importedData.prompts.length} prompts!`, 'success');
   551→                    
   552→                    // Reset file input
   553→                    event.target.value = '';
   554→                }
   555→            );
   556→        } catch (error) {
   557→            console.error('Import error:', error);
   558→            showToast('Error importing file. Please check the file format.', 'error');
   559→        }
   560→    };
   561→    
   562→    reader.readAsText(file);
   563→}

   588→// Prompt Rendering
   589→function renderPrompts() {
   590→    if (!elements.promptsContainer) return;
   591→    
   592→    // Get filtered and sorted prompts
   593→    const filteredPrompts = getFilteredPrompts();
   594→    const sortedPrompts = sortPrompts(filteredPrompts);
   595→    const paginatedPrompts = paginatePrompts(sortedPrompts);
   596→    
   597→    // Clear container
   598→    elements.promptsContainer.innerHTML = '';
   599→    
   600→    if (paginatedPrompts.length === 0) {
   601→        showEmptyState();
   602→        return;
   603→    }
   604→    
   605→    // Create document fragment for better performance
   606→    const fragment = document.createDocumentFragment();
   607→    
   608→    // Render each prompt
   609→    paginatedPrompts.forEach(prompt => {
   610→        const promptElement = createPromptElement(prompt);
   611→        fragment.appendChild(promptElement);
   612→    });
   613→    
   614→    // Append to container
   615→    elements.promptsContainer.appendChild(fragment);
   616→    
   617→    // Update pagination
   618→    updatePagination(filteredPrompts.length);
   619→    
   620→    // Update showing count
   621→    updateShowingCount(paginatedPrompts.length, filteredPrompts.length);
   622→}

   624→function createPromptElement(prompt) {
   625→    const element = document.createElement('div');
   626→    element.className = `prompt-card ${state.currentView}-view`;
   627→    element.dataset.id = prompt.id;
   628→    
   629→    // Make draggable
   630→    element.setAttribute('draggable', 'true');
   631→

  1697→function updateSearchSuggestions() {
  1698→    if (!elements.searchSuggestions) return;
  1699→    
  1700→    const query = state.searchQuery.toLowerCase();
  1701→    if (query.length < 2) {
  1702→        elements.searchSuggestions.classList.remove('active');
  1703→        return;
  1704→    }
  1705→    
  1706→    const suggestions = new Set();
  1707→    
  1708→    // Search in prompts
  1709→    state.prompts.forEach(prompt => {
  1710→        if (prompt.title.toLowerCase().includes(query)) {
  1711→            suggestions.add(prompt.title);
  1712→        }
  1713→        if (prompt.tags) {
  1714→            prompt.tags.forEach(tag => {
  1715→                if (tag.toLowerCase().includes(query)) {
  1716→                    suggestions.add(`Tag: ${tag}`);
  1717→                }
  1718→            });
  1719→        }
  1720→        if (prompt.content.toLowerCase().includes(query)) {
  1721→            // Extract matching sentence
  1722→            const sentences = prompt.content.split(/[.!?]+/);
  1723→            const matchingSentence = sentences.find(s => 
  1724→                s.toLowerCase().includes(query)
  1725→            );
  1726→            if (matchingSentence) {
  1727→                suggestions.add(matchingSentence.trim().substring(0, 50) + '...');
  1728→            }
  1729→        }
  1730→    });
  1731→    
  1732→    // Search in categories
  1733→    state.categories.forEach(category => {
  1734→        if (category.name.toLowerCase().includes(query)) {
  1735→            suggestions.add(`Category: ${category.name}`);
  1736→        }
  1737→    });
  1738→    
  1739→    const suggestionsArray = Array.from(suggestions).slice(0, 8);
  1740→    
  1741→    if (suggestionsArray.length > 0) {
  1742→        elements.searchSuggestions.innerHTML = suggestionsArray.map(suggestion => `
  1743→            <div class="suggestion-item">${escapeHtml(suggestion)}</div>
  1744→        `).join('');
  1745→        elements.searchSuggestions.classList.add('active');
  1746→        
  1747→        // Add click listeners
  1748→        elements.searchSuggestions.querySelectorAll('.suggestion-item').forEach(item => {
  1749→            item.addEventListener('click', () => {
  1750→                const text = item.textContent.replace(/^(Tag:|Category:)\s*/, '');
  1751→                elements.searchInput.value = text;
  1752→                handleSearch({ target: elements.searchInput });
  1753→            });
  1754→        });
  1755→    } else {
  1756→        elements.searchSuggestions.classList.remove('active');
  1757→    }
  1758→}

  1760→function navigateSuggestions(direction) {
  1761→    const suggestions = document.getElementById('searchSuggestions');
  1762→    const items = suggestions.querySelectorAll('.suggestion-item');

  1764→    
  1765→    const current = suggestions.querySelector('.suggestion-item.highlighted');
  1766→    let nextIndex = 0;
  1767→    
  1768→    if (current) {
  1769→        const currentIndex = Array.from(items).indexOf(current);
  1770→        nextIndex = currentIndex + direction;
  1771→        
  1772→        // Wrap around
  1773→        if (nextIndex < 0) nextIndex = items.length - 1;
  1774→        if (nextIndex >= items.length) nextIndex = 0;
  1775→        
  1776→        current.classList.remove('highlighted');
  1777→    }
  1778→    
  1779→    items[nextIndex].classList.add('highlighted');
  1780→    items[nextIndex].scrollIntoView({ block: 'nearest' });
  1781→}
